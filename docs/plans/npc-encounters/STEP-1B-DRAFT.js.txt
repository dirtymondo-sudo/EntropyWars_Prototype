/* the temper of a walking NPC: { key, ladder, starts, joins, watch } | null (no temper: story cast, or a race not tagged yet).
   `ch` = a race name or { race, id, room }: a resident's own override (rooms[room].residents[id]) wins over its race. */
function hqNpcTemper(ch) {
    const T = HQ_NPC_TEMPERS, o = (typeof ch === 'string') ? { race: ch } : (ch || {});
    const R = o.room && T.rooms[o.room], own = R && R.residents && o.id ? R.residents[o.id] : null;
    const key = own || T.races[String(o.race || '').toLowerCase()] || null;
    return (key && T.tempers[key]) ? Object.assign({ key }, T.tempers[key]) : null;
}
/* the named group of a race ('emergency' ...) | null */
function hqNpcGroup(race) {
    const G = HQ_NPC_TEMPERS.groups, r = String(race || '').toLowerCase();
    for (const k in G) if (G[k].indexOf(r) >= 0) return k;
    return null;
}
/* do these two races stand up for each other? (the same race, or one named group) */
function hqNpcSameGroup(a, b) {
    a = String(a || '').toLowerCase(); b = String(b || '').toLowerCase(); if (!a || !b) return false;
    if (a === b) return !!HQ_NPC_TEMPERS.sameRace;
    const g = hqNpcGroup(a); return !!g && g === hqNpcGroup(b);
}
/* would `guard` step in for `victim`? Its own group, or (a Guard) a Civilian / Coward of its district, unless a noDefend row says no */
function hqNpcDefends(guard, victim) {
    const tg = hqNpcTemper(guard), tv = hqNpcTemper(victim); if (!tg || !tg.joins) return false;
    const g = String((guard && guard.race) || guard || '').toLowerCase(), v = String((victim && victim.race) || victim || '').toLowerCase();
    if (HQ_NPC_TEMPERS.noDefend.some(r => r[0] === g && r[1] === v)) return false;
    if (hqNpcSameGroup(g, v)) return true;
    return tg.key === 'guard' && !!tv && HQ_NPC_TEMPERS.guardDefends.indexOf(tv.key) >= 0;
}
